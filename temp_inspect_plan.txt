
const { getServices, getGallery } = require('./src/lib/db');

// Mock next/cache so the db script works in standalone node
jest = { mock: () => {} }; 

async function inspect() {
    try {
        console.log("Fetching Services...");
        const services = await getServices();
        console.log(`Found ${services.length} services.`);
        if (services.length > 0) {
            console.log("Sample Service Image:", services[0].image);
            console.log("Sample Service Name:", services[0].name);
        }

        console.log("\nFetching Gallery...");
        const gallery = await getGallery();
        console.log(`Found ${gallery.length} gallery items.`);
        if (gallery.length > 0) {
            console.log("Sample Gallery Image:", gallery[0].image);
            console.log("Sample Gallery Title:", gallery[0].title);
        }
        
    } catch (e) {
        console.error("Error fetching data:", e);
    }
}
